cmake_minimum_required(VERSION 3.16)
project(dashboard_qt VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Wichtig für Qt: Aktiviert die automatische Code-Generierung
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

# Sucht nach den benötigten Qt6-Komponenten
find_package(Qt6 REQUIRED COMPONENTS Widgets Mqtt)

# Definiert die ausführbare Datei und ihre Quellcode-Dateien
add_executable(dashboard_qt
    src/main.cpp
    src/mainwindow.cpp
    include/mainwindow.h
)

target_include_directories(dashboard_qt PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# Verknüpft dein Programm mit der Qt-Widget-Bibliothek
target_link_libraries(dashboard_qt PRIVATE 
    Qt6::Widgets
    Qt6::Mqtt)
